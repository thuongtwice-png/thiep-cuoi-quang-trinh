    <script>
        lucide.createIcons();
        
        const openBtn = document.getElementById('open-btn');
        const coverPage = document.getElementById('cover-page');
        const content = document.getElementById('invitation-content');
        const music = document.getElementById('bg-music');
        const scrollBtn = document.getElementById('toggle-scroll');
        const musicBtn = document.getElementById('toggle-music');
        
        let isAutoScrolling = false;
        let isMusicPlaying = false;
        const scriptURL = 'https://script.google.com/macros/s/AKfycbxEyvwHH1xMBlgYtioaqqSRI6O3PWibkmHecz5DGG0YsmhCl9rZUgsi0NPcTq8vBoQ2/exec';

        // Open Invitation
        openBtn.addEventListener('click', () => {
            coverPage.style.opacity = '0';
            setTimeout(() => {
                coverPage.style.display = 'none';
                content.style.display = 'block';
                startHeartRain();
                initReveal();
            }, 1000);

            // Play Music
            music.play().then(() => {
                isMusicPlaying = true;
                document.getElementById('music-icon').parentElement.classList.add('animate-spin-slow');
            }).catch(err => console.log("Auto-play blocked"));

            // Start Auto Scroll after delay
            setTimeout(() => {
                isAutoScrolling = true;
                updateScrollIcon();
                autoScroll();
            }, 3000);
        });

        function autoScroll() {
            if (isAutoScrolling) {
                window.scrollBy({ top: 1, behavior: 'auto' });
                if ((window.innerHeight + window.scrollY) < document.body.offsetHeight) {
                    requestAnimationFrame(autoScroll);
                } else {
                    isAutoScrolling = false;
                    updateScrollIcon();
                }
            }
        }

        // Stop scroll on user interaction
        const stopScroll = () => { if(isAutoScrolling) { isAutoScrolling = false; updateScrollIcon(); } };
        window.addEventListener('wheel', stopScroll);
        window.addEventListener('touchstart', stopScroll);
        window.addEventListener('mousedown', stopScroll);

        scrollBtn.addEventListener('click', () => {
            isAutoScrolling = !isAutoScrolling;
            updateScrollIcon();
            if(isAutoScrolling) autoScroll();
        });

        function updateScrollIcon() {
            const icon = document.getElementById('scroll-icon');
            icon.setAttribute('data-lucide', isAutoScrolling ? 'pause' : 'play');
            lucide.createIcons();
        }

        musicBtn.addEventListener('click', () => {
            if (isMusicPlaying) {
                music.pause();
                musicBtn.classList.remove('animate-spin-slow');
            } else {
                music.play();
                musicBtn.classList.add('animate-spin-slow');
            }
            isMusicPlaying = !isMusicPlaying;
        });

        document.getElementById('rsvp-form').addEventListener('submit', async (e) => {
            e.preventDefault();
            const btn = document.getElementById('submit-btn');
            btn.innerText = 'ĐANG GỬI...';
            btn.disabled = true;

            const data = {
                name: document.getElementById('guest-name').value,
                attendance: document.getElementById('attendance').value,
                message: document.getElementById('message').value,
                timestamp: new Date().toLocaleString('vi-VN')
            };

            try {
                await fetch(scriptURL, {
                    method: 'POST',
                    mode: 'no-cors',
                    body: JSON.stringify(data)
                });
                document.getElementById('rsvp-container').innerHTML = `
                    <div class="bg-green-50 p-8 rounded-3xl border border-green-100 flex flex-col items-center">
                        <i data-lucide="check-circle" class="text-green-500 w-12 h-12 mb-4"></i>
                        <h4 class="text-green-800 font-bold text-xl">Cảm ơn bạn đã xác nhận!</h4>
                    </div>
                `;
                lucide.createIcons();
            } catch (error) {
                alert('Có lỗi xảy ra, vui lòng thử lại!');
                btn.innerText = 'GỬI XÁC NHẬN';
                btn.disabled = false;
            }
        });

        function startHeartRain() {
            setInterval(() => {
                const heart = document.createElement('div');
                heart.className = 'falling-heart';
                heart.innerHTML = `<i data-lucide="heart" class="text-rose-300 fill-current w-4 h-4"></i>`;
                heart.style.left = Math.random() * 100 + '%';
                heart.style.animationDuration = (Math.random() * 3 + 4) + 's';
                heart.style.opacity = Math.random() * 0.5 + 0.3;
                document.body.appendChild(heart);
                lucide.createIcons();
                setTimeout(() => heart.remove(), 7000);
            }, 800);
        }

        function initReveal() {
            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) entry.target.classList.add('active');
                });
            }, { threshold: 0.1 });
            document.querySelectorAll('.reveal').forEach(el => observer.observe(el));
        }
    </script>
